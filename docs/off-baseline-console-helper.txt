// Paste this once into DevTools Console only for the Extension Off baseline.
// It records route ids, numeric DOM/heap metrics and sampled switch latency.
// It never reads message text/content.
(() => {
  const history = [];
  let lastSwitchLatencyMs = null;

  const conversationId = () => {
    const match = location.pathname.match(/^\/c\/([^/?#]+)/);
    return match?.[1] ?? null;
  };

  const turns = () => Array.from(document.querySelectorAll('[data-testid="conversation-turn"]'));

  const visibleRounds = () => {
    const turnEls = turns();
    if (turnEls.length) {
      const users = turnEls.filter((turn) =>
        turn.matches('[data-message-author-role="user"]') ||
        Boolean(turn.querySelector('[data-message-author-role="user"]'))
      ).length;
      return users || Math.ceil(turnEls.length / 2);
    }
    return document.querySelectorAll('[data-message-author-role="user"]').length;
  };

  const conversationDomNodes = () => turns().reduce(
    (sum, turn) => sum + 1 + turn.querySelectorAll('*').length,
    0
  );

  const heapUsedMb = () => {
    const bytes = performance.memory?.usedJSHeapSize;
    return Number.isFinite(bytes) ? Math.round(bytes / 1024 / 1024 * 10) / 10 : null;
  };

  const snapshot = () => {
    const value = {
      conversationId: conversationId(),
      renderedRounds: visibleRounds(),
      conversationDomNodes: conversationDomNodes(),
      documentDomNodes: 1 + document.documentElement.querySelectorAll('*').length,
      heapUsedMb: heapUsedMb(),
      switchLatencyMs: lastSwitchLatencyMs,
      capturedAt: Date.now()
    };
    history.push({ ...value });
    return value;
  };

  const measureNextSwitch = () => {
    const startUrl = location.href;
    const startedAt = performance.now();
    return new Promise((resolve) => {
      const check = () => {
        if (location.href !== startUrl && turns().length > 0) {
          lastSwitchLatencyMs = Math.round((performance.now() - startedAt) * 10) / 10;
          resolve(lastSwitchLatencyMs);
          return;
        }
        requestAnimationFrame(check);
      };
      requestAnimationFrame(check);
    });
  };

  window.__CSG_BASELINE__ = Object.freeze({
    snapshot,
    history: () => history.map((item) => ({ ...item })),
    clearHistory: () => { history.length = 0; },
    measureNextSwitch
  });

  console.log('CSG baseline helper ready:', window.__CSG_BASELINE__);
})();
